    37→function useDebounced(value: string, delay: number) {
    38→  const [debounced, setDebounced] = useState(value);
    39→  useEffect(() => {
    40→    const id = window.setTimeout(() => setDebounced(value), delay);
    41→    return () => window.clearTimeout(id);
    42→  }, [value, delay]);
    43→  return debounced;
    44→}

   288→function RadarPanel() {
   289→  const [hidePumped, setHidePumped] = useState(true);
   290→  const [group, setGroup] = useState<RadarSignal | "all">("all");
   291→  const radar = useQuery({
   292→    queryKey: ["radar"],
   293→    queryFn: getRadar,
   294→    refetchInterval: 5_000,
   295→  });
   296→
   297→  const rows = radar.data?.rows ?? [];
   298→  const alerts = rows.filter(
   299→    (row) => row.score >= RADAR_ALERT_MIN_SCORE && row.volume24h > RADAR_ALERT_MIN_VOLUME_24H,
   300→  );
   301→  // Group counts are counted after the hide-pumped filter, so the chips
   302→  // always describe the list you are looking at.
   303→  const base = hidePumped ? rows.filter((row) => row.change24h <= 8) : rows;
   304→  const counts = useMemo(() => {
   305→    const out: Record<RadarSignal, number> = { pumping: 0, dumping: 0, long: 0, short: 0, flat: 0 };
   306→    for (const row of base) out[signalOf(row)] += 1;
   307→    return out;
   308→  }, [base]);
   309→  const shown = group === "all" ? base : base.filter((row) => signalOf(row) === group);
   310→  const updated = radar.dataUpdatedAt > 0 ? new Date(radar.dataUpdatedAt) : null;
   311→
   312→  return (
   313→    <div className="mt-4">
   314→      <p className="max-w-2xl text-sm leading-relaxed text-muted">
   315→        Volume spikes and early momentum across the top 40 USDT pairs on Binance, refreshed every 5 seconds.
   316→        Groups are signals, not predictions — “Go long” means buyers dominate a flat coin right now, not that
   317→        the price will rise. Use it to spot coins that already pumped — and skip them — not to chase pumps.
   318→      </p>
   319→      <p className="mt-2 text-xs text-muted" aria-live="polite">
   320→        <span className="mr-2 inline-block size-1.5 rounded-full bg-up align-middle" aria-hidden="true" />

   322→        {updated ? ` · Updated ${updated.toLocaleTimeString("en-GB")}` : ""}
   323→      </p>
   324→
   325→      {alerts.length > 0 ? (
   326→        <div className="mt-4 rounded-2xl border border-line bg-down-soft px-4 py-3" role="alert">
   327→          {alerts.map((row) => (
   328→            <p key={row.pair} className="text-sm font-medium text-down">
   329→              ⚠️ {row.symbol}: Volume {row.spike.toFixed(1)}x spike, {formatPct(row.change5m)} in 5m, Score{" "}
   330→              {row.score}/100 — Pump Starting
   331→            </p>
   332→          ))}
   333→        </div>
   334→      ) : null}
   335→
   336→      <div className="mt-4 flex flex-wrap gap-2">
   337→        <button
   338→          type="button"
   339→          aria-pressed={hidePumped}
   340→          onClick={() => setHidePumped((value) => !value)}
   341→          className={cn(
   342→            "h-11 rounded-full px-4 text-sm",
   343→            hidePumped ? "bg-ink text-bg" : "border border-line bg-surface",
   344→          )}
   345→        >
   346→          Hide already-pumped (24h &gt; +8%)
   347→        </button>
   348→      </div>
   349→
   350→      <div className="mt-3 flex flex-wrap gap-2" role="group" aria-label="Signal group">
   351→        <button
   352→          type="button"
   353→          aria-pressed={group === "all"}
   354→          onClick={() => setGroup("all")}
   355→          className={cn(
   356→            "h-11 rounded-full px-4 text-sm",
   357→            group === "all" ? "bg-ink text-bg" : "border border-line bg-surface",
   358→          )}
   359→        >
   360→          All <span className="num ml-1 text-xs opacity-70">{base.length}</span>
   361→        </button>

   363→          <button
   364→            key={item}
   365→            type="button"
   366→            aria-pressed={group === item}
   367→            onClick={() => setGroup(item)}

   369→              "h-11 rounded-full px-4 text-sm",
   370→              group === item ? "bg-ink text-bg" : "border border-line bg-surface",
   371→            )}
   372→          >
   373→            {radarSignalLabel[item]}{" "}

   375→          </button>
   376→        ))}
   377→      </div>

   379→      {radar.isError ? (
   380→        <div className="mt-4 rounded-2xl border border-line bg-surface p-8">
   381→          <h2 className="font-display text-2xl">The radar didn’t load</h2>
   382→          <p className="mt-2 text-sm text-muted">
   383→            {radar.error instanceof Error ? radar.error.message : "Try again."}
   384→          </p>
   385→          <button
   386→            type="button"
   387→            onClick={() => void radar.refetch()}
   388→            className="mt-6 h-11 rounded-full bg-ink px-5 text-sm font-medium text-bg"
   389→          >
   390→            Try again
   391→          </button>
   392→        </div>

   413→        now” is 3%+ below the session high with a volume spike and the price falling. “Go long” is buyers
   414→        taking 70%+ of volume on a flat coin (24h −5% to +8%, last 1m and 5m within ±1%) with volume rising
   415→        2x or more — “Go short” is the mirror image, sellers taking 70%+. “Early” means a 3x volume spike
   416→        while the day is still flat. “Volume leading” means the coin sat flat for 2–3 minutes while volume
   417→        spiked — the money arrived before any price move.
   418→      </p>

   423→export function Board({ initial }: { initial: Board }) {
   424→  const router = useRouter();
   425→  const [tab, setTab] = useState<TabId>("rising");
   426→  const [windowKey, setWindowKey] = useState<WindowKey>(initial.window);
   427→  const [book, setBook] = useState<BookKey>(initial.book);
   428→  const [kind, setKind] = useState<MoverKind>("all");
   429→  const [query, setQuery] = useState("");
   430→  const [sort, setSort] = useState<"rank" | "change24h" | "change7d" | "volume24h" | "marketCap">("rank");
   431→  const debounced = useDebounced(query, 220);
   432→  const savedIds = useWatchlist((state) => state.ids);
   433→
   434→  const board = useQuery({
   435→    queryKey: ["board", windowKey, book],
   436→    queryFn: () => getBoard({ data: { window: windowKey, book } }),
   437→    initialData: windowKey === initial.window && book === initial.book ? initial : undefined,
   438→    initialDataUpdatedAt:
   439→      windowKey === initial.window && book === initial.book ? Date.parse(initial.updatedAt) : undefined,
   440→    refetchInterval: 60_000,
   441→  });
   442→
   443→  const saved = useQuery({
   444→    queryKey: ["saved", savedIds],
   445→    queryFn: () => getSaved({ data: { ids: savedIds } }),
   446→    enabled: tab === "saved" && savedIds.length > 0 && debounced.trim().length < 2,
   447→    refetchInterval: tab === "saved" ? 60_000 : false,
   448→  });
   449→
   450→  const remote = useQuery({
   451→    queryKey: ["search", debounced],
   452→    queryFn: () => searchCoins({ data: { q: debounced } }),
   453→    enabled: debounced.trim().length >= 2,
   454→  });
   455→
   456→  const data = board.data;
   457→  const shownWindow = data?.window ?? windowKey;
   458→
   459→  const searching = debounced.trim().length >= 2;
   460→
   461→  const list = useMemo(() => {
   462→    if (!data) return [] as Coin[];
   463→    const q = query.trim().toLowerCase();
   464→    const filterQuery = (rows: Coin[]) => {
   465→      if (!q || (searching && remote.isFetched && !remote.isError)) return rows;
   466→      return rows.filter(
   467→        (coin) => coin.symbol.toLowerCase().includes(q) || coin.name.toLowerCase().includes(q),
   468→      );
   469→    };
   470→
   471→    if (searching && remote.isFetched && !remote.isError) return remote.data ?? [];
   472→
   473→    if (tab === "radar") return [];
   474→
   475→    if (tab === "markets") {
   476→      const rows = [...data.markets];
   477→      rows.sort((a, b) => {
   478→        if (sort === "rank") return a.rank - b.rank;
   479→        return b[sort] - a[sort];
   480→      });
   481→      return filterQuery(rows);
   482→    }
   483→    if (tab === "saved") return filterQuery(saved.data ?? []);
   484→    const source = tab === "rising" ? data.rising : data.falling;
   485→    const filtered =
   486→      kind === "all"
   487→        ? source
   488→        : source.filter((coin) => signalFor(changeOf(coin, shownWindow), coin.change24h) === kind);
   489→    return filterQuery(filtered);
   490→  }, [data, kind, query, remote.data, remote.isError, remote.isFetched, saved.data, searching, shownWindow, sort, tab]);
   491→
   492→  if (board.isError) {
   493→    return (
   494→      <Shell>
   495→        <div className="rounded-2xl border border-line bg-surface p-8">
   496→          <h1 className="font-display text-3xl">The tape didn’t load</h1>
   497→          <p className="mt-2 max-w-lg text-muted">{board.error instanceof Error ? board.error.message : "Try again."}</p>
   498→          <button
   499→            type="button"
   500→            onClick={() => void router.invalidate()}
   501→            className="mt-6 h-11 rounded-full bg-ink px-5 text-sm font-medium text-bg"
   502→          >
   503→            Try again
   504→          </button>
   505→        </div>
   506→      </Shell>
   507→    );
   508→  }
   509→
   510→  if (!data) {
   511→    return (
   512→      <Shell>
   513→        <p className="font-display text-3xl">Pulling the tape…</p>
   514→      </Shell>
   515→    );
   516→  }
   517→
   518→  const moverTab = tab === "rising" || tab === "falling";
   519→  const lead = !searching && moverTab ? list[0] : undefined;
   520→  const rows = lead ? list.slice(1) : list;
   521→  const quiet =
   522→    moverTab &&
   523→    !searching &&
   524→    list.length > 0 &&
   525→    list.every((coin) => Math.abs(changeOf(coin, shownWindow)) < 0.15);
   526→
   527→  const kindOptions: Array<{ id: MoverKind; label: string }> =
   528→    tab === "falling"
   529→      ? [
   530→          { id: "all", label: "All" },
   531→          { id: "fade", label: "Fades" },
   532→          { id: "slide", label: "Slides" },
   533→        ]
   534→      : [
   535→          { id: "all", label: "All" },
   536→          { id: "bounce", label: "Bounces" },
   537→          { id: "run", label: "Runs" },
   538→        ];
   539→
   540→  return (
   541→    <Shell updatedAt={data.updatedAt}>
   542→      <div className="max-w-2xl">
   543→        <h1 className="font-display text-4xl leading-tight tracking-tight sm:text-5xl">
   544→          What’s moving before it isn’t.
   545→        </h1>
   546→        <p className="mt-3 text-base leading-relaxed text-muted">
   547→          Rising and falling tabs sort coins by the last few minutes, not by yesterday’s headline. Open one for
   548→          the full price history.
   549→        </p>
   550→      </div>
   551→
   552→      <section className="mt-6 grid grid-cols-2 gap-3 lg:grid-cols-4" aria-label="Market snapshot">
   553→        <Stat label="Market cap" value={formatCompact(data.global.marketCap)} change={data.global.marketCapChange24h} />
   554→        <Stat label="24h volume" value={formatCompact(data.global.volume24h)} change={data.global.volumeChange24h} />
   555→        <Stat label="Bitcoin share" value={`${data.global.btcDominance.toFixed(1)}%`} />
   556→        <Stat label="Coins listed" value={new Intl.NumberFormat("en-US").format(data.global.listed)} />
   557→      </section>
   558→
   559→      <div className="mt-6 flex flex-col gap-3 sm:flex-row sm:items-center">
   560→        <label className="relative block min-w-0 flex-1">
   561→          <span className="sr-only">Search coins</span>
   562→          <Search className="pointer-events-none absolute top-1/2 left-3 size-4 -translate-y-1/2 text-muted" />
   563→          <input
   564→            value={query}
   565→            onChange={(event) => setQuery(event.target.value)}
   566→            placeholder="Search any coin"
   567→            className="h-11 w-full rounded-full border border-line bg-surface pr-4 pl-10 text-ink outline-none placeholder:text-muted"
   568→          />
   569→        </label>
   570→        <button
   571→          type="button"
   572→          onClick={() => void board.refetch()}
   573→          className="inline-flex h-11 items-center justify-center gap-2 rounded-full border border-line bg-surface px-4 text-sm"
   574→        >
   575→          <RefreshCw className={cn("size-4", board.isFetching && "animate-spin")} />
   576→          Refresh
   577→        </button>
   578→      </div>
   579→
   580→      <div className="mt-5 flex flex-wrap gap-2" role="tablist" aria-label="Tape">
   581→        <Pill active={tab === "markets" && !searching} onClick={() => setTab("markets")}>
   582→          Markets
   583→        </Pill>
   584→        <Pill active={tab === "rising" && !searching} onClick={() => setTab("rising")}>
   585→          Rising
   586→          {data.risingCount > 0 ? <span className="num ml-2 text-xs opacity-70">{data.risingCount}</span> : null}
   587→        </Pill>
   588→        <Pill active={tab === "falling" && !searching} onClick={() => setTab("falling")}>
   589→          Falling
   590→          {data.fallingCount > 0 ? <span className="num ml-2 text-xs opacity-70">{data.fallingCount}</span> : null}
   591→        </Pill>
   592→        <Pill active={tab === "radar" && !searching} onClick={() => setTab("radar")}>
   593→          Radar
   594→        </Pill>
   595→        <Pill active={tab === "saved" && !searching} onClick={() => setTab("saved")}>
   596→          Saved
   597→          {savedIds.length > 0 ? <span className="num ml-2 text-xs opacity-70">{savedIds.length}</span> : null}
   598→        </Pill>
   599→      </div>
   600→
   601→      {searching ? (
   602→        <p className="mt-4 text-sm text-muted">
   603→          {remote.isFetching ? "Searching the full list…" : `${list.length} match${list.length === 1 ? "" : "es"} across listed coins.`}
   604→        </p>
   605→      ) : moverTab ? (
   606→        <div className="mt-4 flex flex-col gap-3">
   607→          <div className="flex flex-wrap gap-2" role="group" aria-label="Time window">
   608→            {windows.map((item) => (
   609→              <button
   610→                key={item}
   611→                type="button"
   612→                aria-pressed={windowKey === item}
   613→                onClick={() => setWindowKey(item)}
   614→                className={cn(
   615→                  "h-11 rounded-full px-4 text-sm",
   616→                  windowKey === item ? "bg-accent text-bg" : "border border-line bg-surface",
   617→                )}
   618→              >
   619→                Last {windowLabel[item]}
   620→              </button>
   621→            ))}
   622→          </div>
   623→          <div className="flex flex-wrap gap-2" role="group" aria-label="Liquidity">
   624→            <button
   625→              type="button"
   626→              aria-pressed={book === "tradable"}
   627→              onClick={() => setBook("tradable")}
   628→              className={cn(
   629→                "h-11 rounded-full px-4 text-sm",
   630→                book === "tradable" ? "bg-ink text-bg" : "border border-line bg-surface",
   631→              )}
   632→            >
   633→              Tradable books
   634→            </button>
   635→            <button
   636→              type="button"
   637→              aria-pressed={book === "deep"}
   638→              onClick={() => setBook("deep")}
   639→              className={cn(
   640→                "h-11 rounded-full px-4 text-sm",
   641→                book === "deep" ? "bg-ink text-bg" : "border border-line bg-surface",
   642→              )}
   643→            >
   644→              Deep books
   645→            </button>
   646→          </div>
   647→          <div className="flex flex-wrap gap-2">
   648→            {kindOptions.map((option) => (
   649→              <button
   650→                key={option.id}
   651→                type="button"
   652→                aria-pressed={kind === option.id}
   653→                onClick={() => setKind(option.id)}
   654→                className={cn("h-11 rounded-full px-3 text-sm", kind === option.id ? "bg-bg-deep text-ink" : "text-muted")}
   655→              >
   656→                {option.label}
   657→              </button>
   658→            ))}
   659→          </div>
   660→          <p className="max-w-2xl text-sm leading-relaxed text-muted">
   661→            {tab === "rising"
   662→              ? `Sorted by the last ${windowLabel[shownWindow]}. A bounce is up now but still red today — often the earliest tell. A run is already green on the day.`
   663→              : `Sorted by the last ${windowLabel[shownWindow]}. A fade is slipping after a green day. A slide is down on both.`}{" "}
   664→            {book === "deep"
   665→              ? "Deep books are names above $250M with at least $5M traded today."
   666→              : "Tradable books are names above $20M with at least $500k traded today, so a spike is more likely to be a real market."}
   667→          </p>
   668→          {data.window !== windowKey || data.book !== book ? (
   669→            <p className="text-sm text-accent">Switching the tape…</p>
   670→          ) : null}
   671→          {quiet ? (
   672→            <p className="text-sm text-muted">Quiet window — these are the largest ticks, all under 0.15%.</p>
   673→          ) : null}
   674→        </div>
   675→      ) : tab === "markets" ? (
   676→        <div className="mt-4 flex flex-wrap gap-2">
   677→          {(
   678→            [
   679→              ["rank", "Rank"],
   680→              ["change24h", "24h"],
   681→              ["change7d", "7d"],
   682→              ["marketCap", "Cap"],
   683→              ["volume24h", "Volume"],
   684→            ] as const
   685→          ).map(([id, label]) => (
   686→            <button
   687→              key={id}
   688→              type="button"
   689→              aria-pressed={sort === id}
   690→              onClick={() => setSort(id)}
   691→              className={cn("h-11 rounded-full px-3 text-sm", sort === id ? "bg-bg-deep text-ink" : "text-muted")}
   692→            >
   693→              {label}
   694→            </button>
   695→          ))}
   696→        </div>
   697→      ) : tab === "radar" ? null : (
   698→        <p className="mt-4 text-sm text-muted">Stars stay on this device. Prices refresh with the tape.</p>
   699→      )}
   700→
   701→      {lead && (tab === "rising" || tab === "falling") ? (
   702→        <div className="mt-5">
   703→          <LeadCard coin={lead} window={shownWindow} />
   704→        </div>
   705→      ) : null}
   706→
   707→      {tab === "radar" && !searching ? (
   708→        <RadarPanel />
   709→      ) : (
   710→      <section className="mt-4" aria-label={searching ? "Search results" : "Coins"}>
   711→        {tab === "saved" && savedIds.length === 0 && !searching ? (
   712→          <div className="rounded-2xl border border-dashed border-line px-5 py-10 text-center">
   713→            <p className="font-display text-2xl">Nothing saved yet</p>
   714→            <p className="mt-2 text-sm text-muted">Star a coin on the rising or falling tape to keep it here.</p>
   715→          </div>
   716→        ) : (searching && remote.isFetching && !remote.data) || (tab === "saved" && saved.isLoading) ? (
   717→          <p className="px-1 py-8 text-sm text-muted">Checking prices…</p>
   718→        ) : list.length === 0 ? (
   719→          <div className="rounded-2xl border border-dashed border-line px-5 py-10 text-center">
   720→            <p className="font-display text-2xl">
   721→              {searching ? "No coin by that name" : moverTab ? "Nothing is moving in this cut" : "No rows"}
   722→            </p>
   723→            <p className="mt-2 text-sm text-muted">
   724→              {moverTab && book === "deep"
   725→                ? "No large name has moved in this window. Try tradable books, or a wider window."
   726→                : searching
   727→                  ? "Check the spelling, or browse Rising."
   728→                  : "Pull to refresh in a moment."}
   729→            </p>
   730→          </div>
   731→        ) : (
   732→          <ol className="rounded-2xl border border-line bg-surface px-3 sm:px-4">
   733→            {(lead ? rows : list).map((coin) => {
   734→              const move = changeOf(coin, shownWindow);
   735→              const primary = searching || tab === "markets" || tab === "saved" ? coin.change24h : move;
   736→              const secondary = searching || tab === "markets" || tab === "saved" ? coin.change7d : coin.change24h;
   737→              const secondaryLabel = searching || tab === "markets" || tab === "saved" ? "7d" : "24h";
   738→              const resolved = !searching && moverTab ? signalFor(move, coin.change24h) : undefined;
   739→              return (
   740→                <CoinRow
   741→                  key={coin.id}
   742→                  coin={coin}
   743→                  primary={primary}
   744→                  secondary={secondary}
   745→                  secondaryLabel={secondaryLabel}
   746→                  signal={resolved}
   747→                />
   748→              );
   749→            })}
   750→          </ol>
   751→        )}
   752→        {!searching && moverTab && data.risingCount + data.fallingCount > 0 ? (
   753→          <p className="mt-3 text-xs text-muted">
   754→            Counts are coins that moved at least 0.15% in the last {windowShort[shownWindow]}. The list shows the
   755→            largest of those.
   756→          </p>
   757→        ) : null}
   758→      </section>
   759→      )}
   760→    </Shell>
   761→  );
   762→}
