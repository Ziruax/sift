    38→    38→  const [debounced, setDebounced] = useState(value);
    39→    39→  useEffect(() => {
    40→    40→    const id = window.setTimeout(() => setDebounced(value), delay);

    42→    42→  }, [value, delay]);
    43→    43→  return debounced;
