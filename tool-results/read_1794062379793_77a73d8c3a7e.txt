     1→// Sift radar service — the always-on scanner.
     2→//
     3→// Rebuild of the original Sift Pump Radar with the reliability fixes that
     4→// matter for capturing real opportunities:
     5→//
     6→//   1. It scans server-side every 5 seconds whether or not any browser is
     7→//      open, so the signal log accumulates 24/7 instead of only while one
     8→//      user keeps the tab focused.
     9→//   2. One scanner serves every client over socket.io — N visitors cost the
    10→//      Binance API the same as one, and nobody gets rate-limited.
    11→//   3. Session state (highs, spikes) lives here, so reloads no longer reset
    12→//      the dump detector.
    13→//   4. Every signal is graded: 5/15/60 minutes later the scanner records what
    14→//      the price actually did, plus running MFE/MAE, persisted via the Next.js
    15→//      ingest webhook. Accountability instead of vibes.
    16→//   5. Scoring adds a trade-count spike (filters one-whale-print volume) and
    17→//      candle close strength on top of the original RVOL/momentum/buy-pressure
    18→//      model.
    19→//
    20→// Data: Binance public market-data domain, no key. Polling, not websockets —
    21→// 5-second granularity, honestly labeled as such.
    22→

   140→// Weight-aware Binance fetch with host failover. Every response carries
   141→// x-mbx-used-weight-1m — the shared-IP window usage — which drives the
   142→// adaptive pacing. A 418/429 answer sets a hard, doubling quiet period.
   143→async function binanceFetch(path: string, timeoutMs = 10_000): Promise<unknown> {
   144→  trackRequest();
   145→  if (bannedUntil > Date.now()) throw new Error("banned");
   146→  let lastError: unknown = null;
   147→  for (let attempt = 0; attempt < HOSTS.length; attempt += 1) {
   148→    const host = HOSTS[attempt];
   149→    try {
   150→      const response = await fetch(host + path, {
   151→        headers: { accept: "application/json" },
   152→        signal: AbortSignal.timeout(timeoutMs),
   153→      });
   154→      const weight = Number(response.headers.get("x-mbx-used-weight-1m"));
   155→      if (Number.isFinite(weight) && weight >= 0) lastUsedWeight = weight;
   156→      if (response.status === 418 || response.status === 429) {
   157→        bannedUntil = Date.now() + banBackoffMs;
   158→        banBackoffMs = Math.min(BAN_BACKOFF_MAX_MS, banBackoffMs * 2);
   159→        lastCleanAt = Date.now(); // the quiet period counts as clean time
   160→        throw new Error(`Binance responded ${response.status} — backing off ${Math.round(banBackoffMs / 60000)}min`);
   161→      }
   162→      if (!response.ok) throw new Error(`Binance responded ${response.status}`);
   163→      const data = (await response.json()) as unknown;
   164→      if (Date.now() - lastCleanAt > BAN_CLEAN_RESET_MS) banBackoffMs = BAN_BACKOFF_INITIAL_MS;
   165→      lastCleanAt = Date.now();
   166→      return data;
   167→    } catch (error) {
   168→      lastError = error;
   169→      if (error instanceof Error && error.message.startsWith("Binance responded 4")) throw error;
   170→      // Network errors/timeouts: try the next host.
   171→    }
   172→  }
   173→  throw lastError ?? new Error("Binance unreachable");
   174→}

   357→// ── Scoring ──────────────────────────────────────────────────────────────────
   358→
   359→// Opportunity score, 0–100 — improved from the original three-factor model:
   360→//   RVOL spike      up to 30 pts (1× → 0, 5×+ → 30)
   361→//   Trade spike     up to 10 pts — a volume spike without a trade spike is
   362→//                   one whale print, not a crowd. This filters it.
   363→//   Momentum        up to 25 pts (all three timeframes green +10, 5m ≥ 15m
   364→//                   acceleration +8, 1m ≥ 5m still-accelerating +7)
   365→//   Buy pressure    up to 20 pts (taker-buy share, 50% → 0, 80%+ → 20)
   366→//   Close strength  up to 15 pts — closes near candle highs mean buyers are
   367→//                   paying up into strength, not fading it.
   368→function pumpScore(
   369→  rvol: number,
   370→  tradeSpike: number,
   371→  change1m: number,
   372→  change5m: number,
   373→  change15m: number,
   374→  buyPct: number,
   375→  closeStrength: number,
   376→): number {
   377→  const clamp01 = (v: number) => Math.min(1, Math.max(0, v));
   378→  const rvolPts = 30 * clamp01((rvol - 1) / 4);
   379→  const tradePts = 10 * clamp01((tradeSpike - 1) / 4);
   380→  let momentumPts = 0;
   381→  if (change1m > 0 && change5m > 0 && change15m > 0) momentumPts += 10;
   382→  if (change5m > 0 && change5m >= change15m) momentumPts += 8;
   383→  if (change1m > 0 && change1m >= change5m) momentumPts += 7;
   384→  const buyPts = 20 * clamp01((buyPct - 50) / 30);
   385→  const closePts = 15 * clamp01(closeStrength);
   386→  return Math.round(Math.min(100, Math.max(0, rvolPts + tradePts + momentumPts + buyPts + closePts)));
   387→}
   388→
   389→function badgeFor(score: number, rvol: number, change24h: number, belowHighPct: number): RadarBadge | null {
   390→  if (change24h > 8) return "pumped";
   391→  if (score >= 80) return "pump";
   392→  if (belowHighPct >= 3) return "dumping";
   393→  if (rvol >= 3 && change24h >= -5 && change24h <= 8) return "early";
   394→  return null;
   395→}

   397→function decideSignal(row: RadarRow): RadarSignal {
   398→  if (row.change24h <= 8 && row.score >= 80 && row.change5m > 0) return "pumping";
   399→  if (row.badge === "dumping" && row.rvol >= 2 && row.change5m < 0) return "dumping";
   400→  const flatDay = row.change24h >= -5 && row.change24h <= 8;
   401→  const flatNow = Math.abs(row.change1m) <= 1 && Math.abs(row.change5m) <= 1;
   402→  if (flatDay && flatNow && row.rvol >= 2 && row.buyPct >= 70) return "long";
   403→  if (flatDay && flatNow && row.rvol >= 2 && row.buyPct <= 30) return "short";
   404→  return "flat";
   405→}

   407→// ── Row construction from a ticker + klines ──────────────────────────────────

   409→function toRow(ticker: Ticker24h, klines: Kline[]): RadarRow | null {
   410→  const pair = typeof ticker.symbol === "string" ? ticker.symbol.toUpperCase() : "";
   411→  if (!pair || !pair.endsWith("USDT") || klines.length < 16) return null;
   412→  const closes = klines.map((k) => num(k[4]));
   413→  const quotes = klines.map((k) => num(k[7]));
   414→  const trades = klines.map((k) => num(k[8]));
   415→  const takerBuys = klines.map((k) => num(k[10]));
   416→  const last = closes.length - 1;
   417→
   418→  const move = (from: number, to: number) => {
   419→    const start = closes[from] ?? 0;
   420→    const end = closes[to] ?? 0;
   421→    return start === 0 ? 0 : ((end - start) / start) * 100;
   422→  };
   423→  const change1m = move(last - 1, last);
   424→  const change5m = move(last - 5, last);
   425→  const change15m = move(last - 15, last);
   426→
   427→  // Twelve 5-minute blocks in the last hour = the baseline.
   428→  const blocks: number[] = [];
   429→  const tradeBlocks: number[] = [];
   430→  for (let start = 0; start + 5 <= quotes.length; start += 5) {
   431→    blocks.push(quotes.slice(start, start + 5).reduce((a, b) => a + b, 0));
   432→    tradeBlocks.push(trades.slice(start, start + 5).reduce((a, b) => a + b, 0));
   433→  }
   434→  const volAvg5m = blocks.length > 0 ? blocks.reduce((a, b) => a + b, 0) / blocks.length : 0;
   435→  const tradeAvg5m = tradeBlocks.length > 0 ? tradeBlocks.reduce((a, b) => a + b, 0) / tradeBlocks.length : 0;
   436→  const vol5m = quotes.slice(-5).reduce((a, b) => a + b, 0);
   437→  const trades5m = trades.slice(-5).reduce((a, b) => a + b, 0);
   438→  const rvol = volAvg5m > 0 ? vol5m / volAvg5m : 0;
   439→  const tradeSpike = tradeAvg5m > 0 ? trades5m / tradeAvg5m : 0;
   440→
   441→  const buyQuote5m = takerBuys.slice(-5).reduce((a, b) => a + b, 0);
   442→  const buyPct = vol5m > 0 ? (buyQuote5m / vol5m) * 100 : 50;
   443→
   444→  // Close strength: average (close − low) / (high − low) over the last 5
   445→  // candles. High == low (flat candle) counts as neutral 0.5.
   446→  let strengthSum = 0;
   447→  let strengthCount = 0;
   448→  for (const k of klines.slice(-5)) {
   449→    const high = num(k[2]);
   450→    const low = num(k[3]);
   451→    const close = num(k[4]);
   452→    if (high > low) {
   453→      strengthSum += (close - low) / (high - low);
   454→      strengthCount += 1;
   455→    }
   456→  }
   457→  const closeStrength = strengthCount > 0 ? strengthSum / strengthCount : 0.5;
   458→
   459→  const price = closes.length > 0 ? closes[closes.length - 1] ?? 0 : num(ticker.lastPrice);
   460→  if (price <= 0) return null;
   461→  const sessionHigh = Math.max(sessionHighs.get(pair) ?? 0, ...closes, price);
   462→  sessionHighs.set(pair, sessionHigh);
   463→  const belowHighPct = sessionHigh > 0 ? ((sessionHigh - price) / sessionHigh) * 100 : 0;
   464→
   465→  const volumeLead = rvol >= 3 && Math.abs(change5m) <= 1 && Math.abs(change1m) <= 1;
   466→  const prevRvol = lastRvol.get(pair) ?? null;
   467→  lastRvol.set(pair, rvol);
   468→  const spikeFalling = prevRvol !== null && prevRvol >= 3 && rvol < 1.5;
   469→
   470→  const score = pumpScore(rvol, tradeSpike, change1m, change5m, change15m, buyPct, closeStrength);
   471→  const change24h = num(ticker.priceChangePercent);
   472→  const badge = badgeFor(score, rvol, change24h, belowHighPct);
   473→
   474→  const row: RadarRow = {
   475→    pair,
   476→    base: pair.slice(0, -4),
   477→    price,
   478→    change1m,
   479→    change5m,
   480→    change15m,

   503→// ── Data acquisition — three paced background tasks ─────────────────────────
   504→//
   505→// The old design fired ~60 requests in a burst every 5 seconds; on this
   506→// sandbox's shared egress IP that bursts straight into Binance's rate-limit
   507→// ban. The replacement paces requests one at a time, adapts the pace to the
   508→// shared weight window reported by every response, and splits concerns:
   509→//
   510→//   • klinePump — one pair's candles per pace-tick, rotating through the
   511→//     universe + subscribed extras. No bursts, ever.
   512→//   • statsTask — batched 24h ticker stats once a minute (weight-cheap).
   513→//   • universeTask — full ticker every 5 minutes to re-rank the universe.
   514→//
   515→// The scan itself becomes pure computation over the stores: zero network.
   516→
   517→async function refreshUniverse(): Promise<void> {
   518→  const payload = await binanceFetch("/api/v3/ticker/24hr", 20_000);
   519→  if (!Array.isArray(payload)) throw new Error("Unexpected Binance payload");
   520→  const rows = (payload as Ticker24h[])
   521→    .filter((row) => {
   522→      const symbol = typeof row.symbol === "string" ? row.symbol.toUpperCase() : "";
   523→      if (!symbol.endsWith("USDT")) return false;
   524→      const base = symbol.slice(0, -4);
   525→      if (!base || STABLE_BASES.has(base) || FIAT_BASES.has(base) || PEGGED_BASES.has(base)) return false;
   526→      if (isBStock(base)) return false;
   527→      if (LEVERAGED_SUFFIXES.some((suffix) => base.endsWith(suffix))) return false;
   528→      return num(row.quoteVolume) > 0;
   529→    })
   530→    .sort((a, b) => num(b.quoteVolume) - num(a.quoteVolume))
   531→    .slice(0, UNIVERSE_LIMIT);
   532→  const nextUniverse: string[] = [];
   533→  const nextStats = new Map<string, Ticker24h>();
   534→  for (const row of rows) {
   535→    const symbol = String(row.symbol).toUpperCase();
   536→    nextUniverse.push(symbol);
   537→    nextStats.set(symbol, row);
   538→  }
   539→  universe = nextUniverse;
   540→  statsUpdatedAt = Date.now();
   541→  // Merge: keep stats for subscribed extras, replace universe stats.
   542→  const merged = new Map<string, Ticker24h>();
   543→  for (const [pair, row] of nextStats) merged.set(pair, row);
   544→  for (const [pair, row] of stats) if (!merged.has(pair)) merged.set(pair, row);
   545→  stats.clear();
   546→  for (const [pair, row] of merged) stats.set(pair, row);
   547→  universeUpdatedAt = Date.now();
   548→  // Drop candles for pairs that left the universe (keep subscribed extras).
   549→  const keep = new Set([...nextUniverse, ...pairRooms.keys()]);
   550→  for (const pair of candles.keys()) {
   551→    if (!keep.has(pair)) {
   552→      candles.delete(pair);
   553→      sessionHighs.delete(pair);
   554→      lastRvol.delete(pair);
   555→    }
   556→  }
   557→}

   559→function pumpPaceMs(): number {
   560→  for (const tier of WEIGHT_TIERS) {
   561→    if (lastUsedWeight < tier.maxWeight) return tier.paceMs;
   562→  }
   563→  return WEIGHT_TIERS[WEIGHT_TIERS.length - 1]?.paceMs ?? 500;
   564→}

   566→async function klinePumpLoop(): Promise<void> {
   567→  let cursor = 0;
   568→  const failureStreaks = new Map<string, number>();
   569→  while (!stopped && isCurrentGeneration()) {
   570→    await sleep(pumpPaceMs());
   571→    if (bannedUntil > Date.now()) continue;
   572→    const rotation = [...universe, ...pairRooms.keys()];
   573→    if (rotation.length === 0) continue;
   574→    const pair = rotation[cursor % rotation.length];
   575→    cursor += 1;
   576→    if (!pair) continue;
   577→    try {
   578→      const payload = await binanceFetch(
   579→        `/api/v3/klines?symbol=${pair}&interval=1m&limit=${KLINE_LIMIT}`,
   580→        10_000,
   581→      );
   582→      if (Array.isArray(payload) && (payload as Kline[]).length >= 16) {
   583→        candles.set(pair, payload as Kline[]);
   584→        failureStreaks.set(pair, 0);
   585→        pairMissing.delete(pair);
   586→      }
   587→    } catch (error) {
   588→      const message = error instanceof Error ? error.message : String(error);
   589→      // "Binance responded 400/404" means the pair genuinely doesn't exist —
   590→      // tell the coin sheet honestly instead of retrying forever.
   591→      if (/responded 4(00|04)/.test(message)) {
   592→        const streak = (failureStreaks.get(pair) ?? 0) + 1;
   593→        failureStreaks.set(pair, streak);
   594→        if (streak >= 2 && pairRooms.has(pair)) {
   595→          const last = pairMissing.get(pair) ?? 0;
   596→          if (Date.now() - last > 60_000) {
   597→            pairMissing.set(pair, Date.now());
   598→            await emitMissingPairUpdate(pair);
   599→          }
   600→        }
   601→      }
   602→      // Bans/network errors: the pace + backoff logic handles recovery.
   603→    }
   604→  }
   605→}

   607→async function statsTaskLoop(): Promise<void> {
   608→  while (!stopped && isCurrentGeneration()) {
   609→    await sleep(STATS_REFRESH_MS);
   610→    if (bannedUntil > Date.now()) continue;
   611→    if (universe.length === 0) continue;
   612→    try {
   613→      // Batched 24h stats for the universe — one request, all pairs.
   614→      const symbols = JSON.stringify(universe);
   615→      const payload = await binanceFetch(
   616→        `/api/v3/ticker/24hr?symbols=${encodeURIComponent(symbols)}`,
   617→        12_000,
   618→      );
   619→      if (Array.isArray(payload)) {
   620→        for (const row of payload as Ticker24h[]) {
   621→          const symbol = typeof row.symbol === "string" ? row.symbol.toUpperCase() : "";
   622→          if (symbol) stats.set(symbol, row);
   623→        }
   624→        statsUpdatedAt = Date.now();
   625→      }
   626→    } catch {
   627→      /* batch failed — stats stay stale, scan keeps running on candles */
   628→    }
   629→    // Subscribed extras get their stats one by one (they are few).
   630→    for (const pair of pairRooms.keys()) {
   631→      if (universe.includes(pair)) continue;
   632→      if (bannedUntil > Date.now()) break;
   633→      try {
   634→        const payload = (await binanceFetch(`/api/v3/ticker/24hr?symbol=${pair}`, 8_000)) as Ticker24h;
   635→        if (typeof payload.symbol === "string") stats.set(pair, payload);
   636→      } catch {
   637→        /* reported as missing by the pump when the symbol is invalid */
   638→      }
   639→    }
   640→  }
   641→}

   643→async function universeTaskLoop(): Promise<void> {
   644→  while (!stopped && isCurrentGeneration()) {
   645→    await sleep(UNIVERSE_REFRESH_MS);
   646→    if (bannedUntil > Date.now()) continue;
   647→    try {
   648→      await refreshUniverse();
   649→    } catch {
   650→      /* keep the previous universe */
   651→    }
   652→  }

   657→async function getAskWall(pair: string, price: number): Promise<{ price: number; valueUsd: number } | null> {
   658→  if (price <= 0) return null;
   659→  const cached = wallCache.get(pair);
   660→  if (cached && Date.now() - cached.at < WALL_TTL_MS) return cached.wall;
   661→  try {
   662→    const payload = (await binanceFetch(`/api/v3/depth?symbol=${pair}&limit=20`, 8_000)) as {
   663→      asks?: Array<[string, string]>;
   664→    };
   665→    const asks = Array.isArray(payload.asks) ? payload.asks : [];
   666→    let best: { price: number; valueUsd: number } | null = null;
   667→    for (const [rawPrice, rawQty] of asks) {
   668→      const askPrice = num(rawPrice);
   669→      const qty = num(rawQty);
   670→      if (askPrice <= 0 || askPrice > price * 1.02) continue;
   671→      const valueUsd = askPrice * qty;
   672→      if (!best || valueUsd > best.valueUsd) best = { price: askPrice, valueUsd };
   673→    }
   674→    wallCache.set(pair, { at: Date.now(), wall: best });
   675→    return best;
   676→  } catch {
   677→    // Feed failed — hide the wall rather than fake one, but retry soon.
   678→    wallCache.set(pair, { at: Date.now() - WALL_TTL_MS + 3_000, wall: null });
   679→    return null;
   680→  }
   681→}

   685→const DIRECTION: Record<SignalType, "long" | "short"> = {
   686→  pump_start: "long",
   687→  volume_lead: "long",
   688→  buyer_pressure: "long",
   689→  seller_pressure: "short",

   694→function signalNote(type: SignalType, row: RadarRow): string {
   695→  switch (type) {
   696→    case "pump_start":
   697→      return `Score ${row.score}: ${row.rvol.toFixed(1)}× volume, takers ${row.buyPct.toFixed(0)}% buys, price ${row.change5m >= 0 ? "+" : ""}${row.change5m.toFixed(2)}% in 5m.`;
   698→    case "volume_lead":
   699→      return `${row.rvol.toFixed(1)}× volume while price held flat (±1%) — the money arrived before the move.`;
   700→    case "buyer_pressure":
   701→      return `Flat coin, takers buying ${row.buyPct.toFixed(0)}% of a ${row.rvol.toFixed(1)}× volume tape.`;
   702→    case "seller_pressure":
   703→      return `Flat coin, takers selling ${(100 - row.buyPct).toFixed(0)}% of a ${row.rvol.toFixed(1)}× volume tape.`;
   704→    case "dump_start":
   705→      return `${row.belowHighPct.toFixed(1)}% below the session high with ${row.rvol.toFixed(1)}× volume and a falling 5m tape.`;
   706→    case "spike_collapse":
   707→      return `Volume spike collapsed to ${row.rvol.toFixed(1)}× — the crowd that pushed the move is leaving.`;
   708→  }
   709→}

   711→function detectSignals(rows: RadarRow[]): TrackedSignal[] {
   712→  const now = Date.now();
   713→  const fired: TrackedSignal[] = [];
   714→  for (const row of rows) {
   715→    const candidates: SignalType[] = [];
   716→    const flatDay = row.change24h >= -5 && row.change24h <= 8;
   717→    const flatNow = Math.abs(row.change1m) <= 1 && Math.abs(row.change5m) <= 1;
   718→
   719→    if (row.score >= 80 && row.change5m > 0 && row.change24h <= 8) candidates.push("pump_start");
   720→    if (row.rvol >= 3 && flatNow && flatDay) candidates.push("volume_lead");
   721→    if (flatDay && flatNow && row.rvol >= 2 && row.buyPct >= 70) candidates.push("buyer_pressure");
   722→    if (flatDay && flatNow && row.rvol >= 2 && row.buyPct <= 30) candidates.push("seller_pressure");
   723→    if (row.belowHighPct >= 3 && row.rvol >= 2 && row.change5m < 0) candidates.push("dump_start");
   724→    if (row.spikeFalling) candidates.push("spike_collapse");
   725→
   726→    for (const type of candidates) {
   727→      const key = `${row.pair}:${type}`;
   728→      const last = cooldowns.get(key) ?? 0;
   729→      if (now - last < SIGNAL_COOLDOWN_MS) continue;
   730→      // Direction-adjusted MFE/MAE run from the moment of the signal.
   731→      fired.push({
   732→        id: `${row.pair}:${type}:${Math.floor(now / 60_000)}`,
   733→        pair: row.pair,
   734→        base: row.base,
   735→        type,
   736→        direction: DIRECTION[type],
   737→        score: row.score,
   738→        price: row.price,
   739→        change24h: row.change24h,
   740→        rvol: row.rvol,
   741→        buyPct: row.buyPct,
   742→        volume24h: row.volume24h,
   743→        note: signalNote(type, row),
   744→        createdAt: now,
   745→        sentNew: false,
   746→        sent5m: false,
   747→        sent15m: false,
   748→        sent60m: false,
   749→        mfe: 0,
   750→        mae: 0,
   751→        final: false,
   752→      });
   753→      cooldowns.set(key, now);
   754→      if (fired.length >= MAX_SIGNALS_PER_SCAN) return fired;
   755→    }
   756→  }
   757→  return fired;
   758→}

   815→async function gradeOutcomes(prices: Map<string, number>): Promise<void> {
   816→  const now = Date.now();
   817→  const updates: Array<Record<string, unknown>> = [];
   818→  const newSends: GradedSignal[] = [];
   819→
   820→  for (const s of trackedSignals as GradedSignal[]) {
   821→    const price = prices.get(s.pair);
   822→    if (price !== undefined && price > 0) {
   823→      const raw = (price / s.price - 1) * 100;
   824→      const favorable = s.direction === "long" ? raw : -raw;
   825→      if (favorable > s.mfe) s.mfe = favorable;
   826→      if (favorable < s.mae) s.mae = favorable;
   827→    }
   828→
   829→    const age = now - s.createdAt;
   830→    const lastPrice = price ?? s.price;
   831→
   832→    if (!s.sentNew) newSends.push(s);
   833→
   834→    if (!s.sent5m && age >= 5 * 60_000) {
   835→      s.price5m = lastPrice;
   836→      s.change5m = (lastPrice / s.price - 1) * 100;
   837→      s.sent5m = true;
   838→    }
   839→    if (!s.sent15m && age >= 15 * 60_000) {
   840→      s.price15m = lastPrice;
   841→      s.change15m = (lastPrice / s.price - 1) * 100;
   842→      s.sent15m = true;
   843→    }
   844→    if (!s.sent60m && age >= OUTCOME_HORIZON_MS) {
   845→      s.price60m = lastPrice;
   846→      s.change60m = (lastPrice / s.price - 1) * 100;
   847→      s.sent60m = true;
   848→      s.final = true;
   849→    }
   850→
   851→    if (s.sent5m || s.sent15m || s.sent60m || s.final) {
   852→      updates.push({
   853→        id: s.id,
   854→        price5m: s.price5m ?? null,
   855→        change5m: s.change5m ?? null,
   856→        price15m: s.price15m ?? null,
   857→        change15m: s.change15m ?? null,
   858→        price60m: s.price60m ?? null,
   859→        change60m: s.change60m ?? null,
   860→        mfe: s.mfe,
   861→        mae: s.mae,
   862→        status: s.final ? "final" : s.sent15m || s.sent5m ? "partial" : "pending",
   863→      });
   864→    }
   865→  }
   866→
   867→  // Persist newly fired signals (retry until the API accepts them).
   868→  for (const s of newSends) {
   869→    const ok = await postIngest({ mode: "new", signal: toApiSignal(s) });
   870→    if (ok) s.sentNew = true;
   871→    else if (Date.now() - s.createdAt > 10 * 60_000) s.sentNew = true; // give up quietly
   872→  }
   873→
   874→  // Batch the outcome updates.
   875→  const matured = updates.filter((u) => u.id !== undefined);
   876→  if (matured.length > 0) {
   877→    await postIngest({ mode: "outcome", updates: matured.slice(0, 200) });
   878→  }
   879→
   880→  // Drop finalized signals older than 2 hours from memory (DB holds history).
   881→  for (let i = trackedSignals.length - 1; i >= 0; i -= 1) {
   882→    const s = trackedSignals[i] as GradedSignalAlias;
   883→    if (s.final && now - s.createdAt > 2 * 60 * 60_000) trackedSignals.splice(i, 1);
   884→  }
   885→  // Hard cap on memory.
   886→  while (trackedSignals.length > 800) trackedSignals.shift();
   887→}

   952→// ── The scan loop — pure computation, zero network ────────────────────────────
   953→
   954→let scanning = false;
   955→let stopped = false;
   956→
   957→async function scan(): Promise<void> {
   958→  if (scanning || stopped || !isCurrentGeneration()) return;
   959→  scanning = true;
   960→  try {
   961→    // The data tasks fill the stores; the scan just reads them. Pairs whose
   962→    // candles haven't arrived yet are skipped, not fatal.
   963→    const pairs = [...new Set([...universe, ...pairRooms.keys()])];
   964→    const rows: RadarRow[] = [];
   965→    const prices = new Map<string, number>();
   966→    for (const pair of pairs) {
   967→      const k = candles.get(pair);
   968→      const ticker = stats.get(pair);
   969→      if (!k || !ticker) continue;
   970→      const row = toRow(ticker, k);
   971→      if (row) {
   972→        rows.push(row);
   973→        prices.set(row.pair, row.price);
   974→      }
   975→    }
   976→    if (rows.length === 0) {
   977→      consecutiveFailures += 1;
   978→      if (consecutiveFailures <= 2 || consecutiveFailures % 12 === 0) {
   979→        console.error(`[scan] no data yet (failures: ${consecutiveFailures})`);
   980→      }
   981→      return;
   982→    }
   983→
   984→    rows.sort((a, b) => b.score - a.score || b.rvol - a.rvol);
   985→    scans += 1;
   986→    consecutiveFailures = 0;
   987→    lastSnapshot = {
   988→      rows,
   989→      updatedAt: new Date().toISOString(),
   990→      universe: universe.length,
   991→      clients: io.engine.clientsCount,
   992→      scans,
   993→      stale: bannedUntil > Date.now(),
   994→    };
   995→    io.emit("radar:snapshot", lastSnapshot);
   996→
   997→    // Per-pair updates for subscribed coin sheets.
   998→    for (const row of rows) {
   999→      if (pairRooms.has(row.pair)) await emitPairUpdate(row);
  1000→    }
  1001→
  1002→    // Signal detection + broadcast.
  1003→    const fired = detectSignals(rows);
  1004→    for (const s of fired) {
  1005→      trackedSignals.push(s);
  1006→      io.emit("signal:new", toApiSignal(s as GradedSignal));
  1007→    }
  1008→    // Outcome grading + persistence.
  1009→    await gradeOutcomes(prices);
  1010→  } finally {
  1011→    scanning = false;
  1012→  }
  1013→}

  1015→function scheduleNext(): void {
  1016→  if (stopped || !isCurrentGeneration()) return;
  1017→  const timer = setTimeout(() => {

  1023→}
